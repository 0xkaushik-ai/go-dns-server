# Example zone file: names this server answers for itself.
#
# Format:   <name>  <ttl seconds>  <type>  <value>
# Types:    A, AAAA, CNAME, MX, TXT
# Every record must be under a $DOMAIN line.

$DOMAIN home.

home.          300  TXT    "my home network"
home.          300  MX     10 mail.home.

router.home.   300  A      192.168.1.1
nas.home.      300  A      192.168.1.20
nas.home.      300  AAAA   fd00::20
mail.home.     300  A      192.168.1.25

# Aliases: files.home and media.home are other names for nas.home.
files.home.    300  CNAME  nas.home.
media.home.    300  CNAME  files.home.

$DOMAIN lab.internal.

web.lab.internal.  60  A  10.10.0.5
db.lab.internal.   60  A  10.10.0.6
